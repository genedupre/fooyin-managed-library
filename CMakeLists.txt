cmake_minimum_required(VERSION 3.25)
project(FooyinManagedLibrary VERSION 0.1.0 LANGUAGES CXX)
if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
  message(FATAL_ERROR "Only 64-bit builds are supported.")
endif()
set(CMAKE_CXX_STANDARD 23)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
if(WIN32)
  find_package(Qt6 6.12 REQUIRED COMPONENTS Core Gui Widgets Sql Network Concurrent Test Svg)
else()
  find_package(Qt6 6.7 REQUIRED COMPONENTS Core Gui Widgets Sql Network Concurrent Test Svg)
endif()
set(FOOYIN_SOURCE "" CACHE PATH "fooyin v0.14.1 source tree")
if(WIN32)
  set(FOOYIN_RUNTIME "C:/Program Files/fooyin" CACHE PATH "Matching installed fooyin")
else()
  set(FOOYIN_RUNTIME "" CACHE PATH "Installed fooyin library directory (contains libfooyin_*.so)")
endif()
file(MAKE_DIRECTORY "${CMAKE_BINARY_DIR}/generated")
foreach(module core gui utils)
  string(TOUPPER "fy${module}" macro)
  if(WIN32)
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/fy${module}_export.h" CONTENT "#pragma once\n#define ${macro}_EXPORT __declspec(dllimport)\n#define ${macro}_NO_EXPORT\n#define ${macro}_DEPRECATED __declspec(deprecated)\n#define ${macro}_DEPRECATED_EXPORT ${macro}_EXPORT ${macro}_DEPRECATED\n")
    add_library(fooyin_${module} SHARED IMPORTED)
    set_target_properties(fooyin_${module} PROPERTIES IMPORTED_IMPLIB "${CMAKE_SOURCE_DIR}/imports/fooyin_${module}.lib" IMPORTED_LOCATION "${FOOYIN_RUNTIME}/fooyin_${module}.dll")
  else()
    file(GENERATE OUTPUT "${CMAKE_BINARY_DIR}/generated/fy${module}_export.h" CONTENT "#pragma once\n#define ${macro}_EXPORT __attribute__((visibility(\"default\")))\n#define ${macro}_NO_EXPORT __attribute__((visibility(\"hidden\")))\n#define ${macro}_DEPRECATED __attribute__((deprecated))\n#define ${macro}_DEPRECATED_EXPORT ${macro}_EXPORT ${macro}_DEPRECATED\n")
    add_library(fooyin_${module} SHARED IMPORTED)
    set_target_properties(fooyin_${module} PROPERTIES IMPORTED_LOCATION "${FOOYIN_RUNTIME}/libfooyin_${module}.so.0.0.0")
  endif()
endforeach()
set(FOOYIN_VERSION "0.14.1-managed.4")
configure_file(fileops/fileops.json.in generated/fileops.json)
file(GLOB sources CONFIGURE_DEPENDS fileops/*.cpp fileops/*.h)
add_library(fyplugin_fileops MODULE ${sources})
set_target_properties(fyplugin_fileops PROPERTIES PREFIX "")
target_include_directories(fyplugin_fileops PRIVATE "${FOOYIN_SOURCE}/include" "${FOOYIN_SOURCE}/src" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(fyplugin_fileops PRIVATE fooyin_core fooyin_gui fooyin_utils Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Sql Qt6::Network Qt6::Concurrent Qt6::Svg)
if(MSVC)
  target_compile_options(fyplugin_fileops PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
else()
  target_compile_options(fyplugin_fileops PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_compile_definitions(fyplugin_fileops PRIVATE QT_NO_KEYWORDS)
enable_testing()
add_executable(managedfiles_test tests/managedfiles_test.cpp fileops/managedfiles.cpp)
target_link_libraries(managedfiles_test PRIVATE Qt6::Core Qt6::Test)
target_include_directories(managedfiles_test PRIVATE fileops)
if(MSVC)
  target_compile_options(managedfiles_test PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
else()
  target_compile_options(managedfiles_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
add_test(NAME managedfiles COMMAND managedfiles_test)
add_executable(plugin_load_test tests/plugin_load_test.cpp)
target_link_libraries(plugin_load_test PRIVATE Qt6::Core)
add_test(NAME plugin_load COMMAND plugin_load_test $<TARGET_FILE:fyplugin_fileops>)
add_executable(integration_test tests/integration_test.cpp)
target_include_directories(integration_test PRIVATE "${FOOYIN_SOURCE}/include" "${FOOYIN_SOURCE}/src" "${CMAKE_BINARY_DIR}/generated")
target_link_libraries(integration_test PRIVATE fooyin_core fooyin_gui fooyin_utils Qt6::Core Qt6::Gui Qt6::Widgets Qt6::Sql Qt6::Network Qt6::Concurrent Qt6::Test)
if(MSVC)
  target_compile_options(integration_test PRIVATE /W4 /permissive- /Zc:__cplusplus /utf-8)
else()
  target_compile_options(integration_test PRIVATE -Wall -Wextra -Wpedantic)
endif()
target_compile_definitions(integration_test PRIVATE FOOYIN_SOURCE="${FOOYIN_SOURCE}")
add_test(NAME integration COMMAND integration_test $<TARGET_FILE:fyplugin_fileops> "${FOOYIN_SOURCE}/tests/data/audio/audiotest.mp3" "${CMAKE_BINARY_DIR}/managed-settings.png")
if(WIN32)
  set_tests_properties(integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_PLUGIN_PATH=${FOOYIN_RUNTIME};QT_QPA_FONTDIR=C:/Windows/Fonts" TIMEOUT 120)
else()
  set_tests_properties(integration PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;LD_LIBRARY_PATH=${FOOYIN_RUNTIME}:$ENV{LD_LIBRARY_PATH}" TIMEOUT 180)
endif()
